\section{Bloch Hamiltonian}
\begin{deftcolorbox}[title = Bloch Hamiltonian]
\begin{equation*}
    \hat{\mathcal{H}}_{\bm{k}} = e^{-i\bm{k}\cdot \hat{\bm{x}}} \hat{\mathcal{H}} e^{i\bm{k}\cdot \hat{\bm{x}}}
\end{equation*}
\end{deftcolorbox}
\begin{theotcolorbox}[title = Blochの定理]
\begin{equation*}
    \hat{\mathcal{H}}_{\bm{k}} \ket{u_n (\bm{k})} = E_n(\bm{k}) \ket{u_n(\bm{k})}
\end{equation*}
\end{theotcolorbox}
